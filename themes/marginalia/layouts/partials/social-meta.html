{{- /* social-meta.html — link previews: Open Graph tags (Facebook, LinkedIn, Slack, iMessage, WhatsApp, Mastodon…) and Twitter/X card tags. Image: the post's `hero` (a bundle resource), saved as JPEG, which every preview service accepts (WebP and SVG aren't safe bets). Never upscaled. • 3:2 and wider heroes are cropped to the 1.91:1 shape of a large preview card, 1200×630 when wide enough. • Everything else keeps its shape, at most 1200px. • 4:3 and wider get X's large card; squarer heroes (album and book covers) get its small square "summary" card instead. Pages with no hero (home, listings) use site.Params.social_image, a path under static/, if one is set, and otherwise get a text-only card. */ -}} {{- $title := cond .IsHome site.Title .Title -}} {{- $desc := partial "page-description.html" . -}} {{- $alt := "" -}} {{- $img := dict -}} {{- with .Params.hero -}} {{- with $.Resources.Get . -}} {{- if in (slice "jpeg" "png" "webp") .MediaType.SubType -}} {{- $src := . -}} {{- if eq .MediaType.SubType "jpeg" }}{{ $src = . | images.Filter images.AutoOrient }}{{ end -}} {{- $out := "" -}} {{- $ratio := div (float $src.Width) $src.Height -}} {{- if ge $ratio 1.5 -}} {{- $w := cond (ge $src.Width 1200) 1200 $src.Width -}} {{- $h := int (math.Round (div (float $w) 1.905)) -}} {{- $out = $src.Fill (printf "%dx%d jpg q80 Center" $w $h) -}} {{- else if or (gt $src.Width 1200) (gt $src.Height 1200) -}} {{- $out = $src.Fit "1200x1200 jpg q80" -}} {{- else -}} {{- $out = $src.Process "jpg q80" -}} {{- end -}} {{- $img = dict "url" $out.Permalink "width" $out.Width "height" $out.Height "large" (ge $ratio 1.3) -}} {{- $alt = $.Params.hero_alt | default $.Title -}} {{- end -}} {{- end -}} {{- end -}} {{- if and (not $img) site.Params.social_image -}} {{- $img = dict "url" (site.Params.social_image | absURL) "large" true -}} {{- $alt = site.Params.social_image_alt | default site.Title -}} {{- end -}} {{ with $desc }}{{ end }} {{- $loc := split (site.Language.Locale | default "en-gb") "-" }} {{- if and .IsPage (eq .Section "posts") }} {{ if gt .Lastmod .Date }}{{ end }} {{ with partial "first-category.html" . }}{{ end }} {{ range first 6 (.Params.tags | default slice) }} {{ end }} {{- else }} {{- end }} {{- with $img }} {{ with .width }}{{ end }} {{ with .height }}{{ end }} {{ with $alt }}{{ end }} {{ with $alt }}{{ end }} {{- else }} {{- end }} {{ with $desc }}{{ end }}
